cmake_minimum_required(VERSION 3.20)
project(token_place_installer_ownership_tests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows path semantics require a Windows test runner")
endif()
add_executable(ownership-tests ownership_tests.cpp)
target_compile_features(ownership-tests PRIVATE cxx_std_17)
target_compile_options(ownership-tests PRIVATE /W4 /WX /EHsc)
set_property(TARGET ownership-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME ownership-policy COMMAND ownership-tests "${CMAKE_CURRENT_BINARY_DIR}/disposable-path-fixture")
