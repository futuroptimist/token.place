cmake_minimum_required(VERSION 3.20)
project(ownership_preflight_fixture_tests LANGUAGES CXX)
enable_testing()
add_executable(ownership_preflight_tests tests.cpp)
target_compile_features(ownership_preflight_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ownership_preflight_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ownership_preflight_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME synthetic_multi_user_ownership COMMAND ownership_preflight_tests)
