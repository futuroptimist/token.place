cmake_minimum_required(VERSION 3.20)
project(token_place_installer_ownership_tests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows path semantics require a Windows test runner")
endif()
add_executable(ownership-tests ownership_tests.cpp)
add_executable(process-fixture process_fixture.cpp)
set_property(TARGET process-fixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Compile production adapters too; local CTest executes only synthetic policy.
add_executable(ownership-probe ownership.cpp)
add_library(ownership-msi SHARED ownership.cpp)
target_compile_definitions(ownership-msi PRIVATE TOKEN_PLACE_MSI)
foreach(adapter ownership-probe ownership-msi)
  target_compile_features(${adapter} PRIVATE cxx_std_17)
  target_compile_options(${adapter} PRIVATE /W4 /WX /EHsc)
  target_link_libraries(${adapter} PRIVATE msi advapi32)
  set_property(TARGET ${adapter} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
target_compile_features(ownership-tests PRIVATE cxx_std_17)
target_compile_options(ownership-tests PRIVATE /W4 /WX /EHsc)
set_property(TARGET ownership-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME ownership-policy COMMAND ownership-tests "${CMAKE_CURRENT_BINARY_DIR}/disposable-path-fixture")
